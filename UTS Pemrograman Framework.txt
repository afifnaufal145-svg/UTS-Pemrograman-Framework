function solution(input) {
  // input: [nama, uts, uas]

  class Mahasiswa {
    constructor(nama, uts, uas) {
      this.nama = nama;
      this.uts = uts;
      this.uas = uas;
    }

    getNilai() {
      return (this.uts + this.uas) / 2;
    }

    getGrade() {
      const nilai = this.getNilai();

      if (nilai >= 85) return 'A';
      else if (nilai >= 75) return 'B';
      else if (nilai >= 65) return 'C';
      else if (nilai >= 55) return 'D';
      else return 'E';
    }

    getInfo() {
      return `Nama: ${this.nama}, Nilai: ${this.getNilai()}, Grade: ${this.getGrade()}`;
    }
  }

  const mhs = new Mahasiswa(input[0], input[1], input[2]);
  return mhs.getInfo();
}
